#!/bin/bash
set -e

# ---------------------------------------- CONFIG ----------------------------------------
USER_NAME="$(whoami)"
BACKEND_DIR="/home/$USER_NAME/projects/backend"
LOG_DIR="/home/$USER_NAME/server_logs"
URL="http://localhost:8000/"
# ----------------------------------------------------------------------------------------

mkdir -p "$LOG_DIR"
echo "===== START $(date) =====" >> "$LOG_DIR/boot.log"

# Make sure system PATH is decent (systemd sometimes strips PATH)
export PATH="/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin"

# ---------- BACKEND: update + start ----------
cd "$BACKEND_DIR"

git fetch origin >> "$LOG_DIR/git.log" 2>&1
git reset --hard "origin/main" >> "$LOG_DIR/git.log" 2>&1

# Activate virtualenv if you use one (uncomment if needed):
# source "$BACKEND_DIR/venv/bin/activate"

# Start Django only if it's not already running
pgrep -f "manage.py runserver" > /dev/null || \
    nohup python3 manage.py runserver 0.0.0.0:8000 > "$LOG_DIR/django.log" 2>&1 &

# ---------- WAIT FOR GUI (X) ----------
# Chromium needs a display; wait up to 60s for X to start
for i in {1..60}; do
    [ -S /tmp/.X11-unix/X0 ] && break
    sleep 1
done

export DISPLAY=:0
export XAUTHORITY="/home/$USER_NAME/.Xauthority"

# ---------- WAIT FOR DJANGO ----------
# Wait up to 120s for Django to be responsive before opening browser
for i in {1..120}; do
    curl -s "$URL" > /dev/null 2>&1 && break
    sleep 1
done

# ---------- OPEN CHROMIUM KIOSK ----------
# Don't open a second copy if already running
pgrep -x chromium-browser > /dev/null || \
    nohup chromium-browser --kiosk --noerrdialogs --disable-infobars "$URL" \
    > "$LOG_DIR/chromium.log" 2>&1 &

echo "===== DONE $(date) =====" >> "$LOG_DIR/boot.log"
